// ============================================================ // JioTV+ — Cloudflare Worker: UI + Stream Proxy // Deploy: npx wrangler deploy worker.js --name jiotvplus-ui // ============================================================ // --- Configuration --- const JSON_API = 'https://jiotvplus.dr-strange.workers.dev/watch/fetch.json'; const M3U_API = 'https://jiotvplus.dr-strange.workers.dev/api/jiotvplus.m3u'; const JIO_CDN = 'https://jiotvpllive.cdn.jio.com'; // ============================================================ // STREAM PROXY — Handle CORS + Auth for MPD & segments // ============================================================ async function handleStreamProxy(request, url) { // Extract the target Jio CDN path // Request comes as: /proxy/bpk-tv/StarSportsHD2Hin_BTS/WDVLive/index.mpd?__hdnea__=... const targetPath = url.pathname.replace('/proxy', ''); const targetUrl = JIO_CDN + targetPath + (url.search || ''); // Build proxy request — forward auth token as cookie const proxyHeaders = new Headers(); // Extract __hdnea__ from query and set as Cookie const hdnea = url.searchParams.get('__hdnea__'); if (hdnea) { proxyHeaders.set('Cookie', `__hdnea__=${hdnea}`); } // Forward User-Agent (Jio CDN may check this) proxyHeaders.set('User-Agent', 'Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:150.0) Gecko/20100101 Firefox/150.0'); proxyHeaders.set('Referer', 'https://www.jio.com/'); proxyHeaders.set('Origin', 'https://www.jio.com'); // Fetch from Jio CDN const response = await fetch(targetUrl, { headers: proxyHeaders, method: request.method }); // Clone response and add CORS headers const corsHeaders = new Headers(response.headers); corsHeaders.set('Access-Control-Allow-Origin', '*'); corsHeaders.set('Access-Control-Allow-Methods', 'GET, OPTIONS'); corsHeaders.set('Access-Control-Allow-Headers', '*'); corsHeaders.set('Access-Control-Allow-Credentials', 'true'); // For MPD manifests, rewrite segment URLs to go through proxy let body = await response.text(); if (targetPath.endsWith('.mpd')) { // Rewrite all segment URLs in the MPD to go through our proxy // MPDs contain , , etc. body = body.replace( /https:\/\/jiotvpllive\.cdn\.jio\.com\//g, `${url.origin}/proxy/` ); // Also rewrite relative URLs const baseDir = targetPath.substring(0, targetPath.lastIndexOf('/') + 1); body = body.replace( /([^<]+)<\/BaseURL>/g, (match, baseUrl) => { if (baseUrl.startsWith('http')) return match; return `${url.origin}/proxy${baseDir}${baseUrl}`; } ); } return new Response(body, { status: response.status, headers: corsHeaders }); } // ============================================================ // DASHBOARD - index.html // ============================================================ const INDEX_HTML = ` JioTV+ LIVE

`; // ============================================================ // WORKER ROUTER // ============================================================ export default { async fetch(request, env, ctx) { const url = new URL(request.url); const path = url.pathname; // Handle CORS preflight if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': '*', 'Access-Control-Max-Age': '86400' } }); } // Stream proxy: /proxy/* -> Jio CDN if (path.startsWith('/proxy/')) { return handleStreamProxy(request, url); } // Player page if (path === '/player' || path.startsWith('/player?')) { return new Response(PLAYER_HTML, { headers: { 'content-type': 'text/html;charset=UTF-8', 'cache-control': 'public, max-age=3600' } }); } // Dashboard (everything else) return new Response(INDEX_HTML, { headers: { 'content-type': 'text/html;charset=UTF-8', 'cache-control': 'public, max-age=3600' } }); } };
